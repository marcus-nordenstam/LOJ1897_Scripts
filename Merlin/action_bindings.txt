# action_bindings.txt
#
# Maps Merlin actions to GRYM resources. Loaded once at startup.
# The Merlin action name [SECTION] is the key.
# Motor comes from the Merlin ontology, not from this file.
#
# Keys:
#   expression = <filename.esp>           expression from library (implies generic expression proc_anim)
#   proc_anim = <name>                    specialized proc_anim (tell, lookat, blink)
#   update = <name>                       specialized update function (walk, run, walkto, turn, etc.)
#   anim = <exact name>                   animation (gender-neutral, without .gra extension)
#   anim.female = <exact name>            female-specific animation
#   anim.male = <exact name>              male-specific animation
#   anim_flags = <flags>                  loop, pingpong, reset, randomize_start, backwards
#   alias = <MERLIN_ACTION>               reuse another action's animation
#   state = <flags>                       state bits to set: walking, running, sitting, standing, telling
#   speed = <float>                       default movement speed or speech rate multiplier
