# kind_bindings.txt
#
# Maps Merlin entity kinds to GRYM asset/type.
# Loaded once at startup by ProximitySpawner.
#
# Keys:
#   type = character | geo | prefab
#   asset = <filename>      (for geo/prefab types)
#
# For type=character, per-instance model comes from SpawnDataTable.
# For type=geo, asset is a .grg geometry file.
# For type=prefab, asset is a .grs prefab file.
#
# NOTE: Non-character kinds are migrating to .phy files (Game/Env/Physical/).
# Only character bindings remain here.

[human]
type = character
